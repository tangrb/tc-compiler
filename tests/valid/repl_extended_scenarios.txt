; REPL 手工场景目录：每个编号须在独立 tc-vm --repl 会话中运行。
; 自动回归以 scripts/vm/run_tests.sh 的 REPL 区块为准。
; 1. 多行变量链式运算
var a: int32 = 10
var b: int32 = 20
var c: int32 = add(int32, a, b)
var d: int32 = mul(int32, c, 2)
writeln(int32, d)
:quit

; 2. 使用 let 常量再做运算
let N: int32 = 100
var x: int32 = add(int32, N, 50)
writeln(int32, x)
:quit

; 3. :reset 后重算
var z: int32 = 99
:reset
var z2: int32 = 11
var z3: int32 = 22
var z4: int32 = add(int32, z2, z3)
writeln(int32, z4)
:quit

; 4. 多进制字面量
var a: uint32 = 0xFF
var b: uint32 = 0b1010
writeln(uint32, add(uint32, a, b))
:quit

; 5. :vars 检查
var p: int8 = 42
var q: uint64 = 123
:vars
:quit

; 6. 类型边界
var a: int8 = 127
var b: int8 = 1
var c: int8 = add(int8, wrap, a, b)
writeln(int8, c)
:quit

; 7. 已初始化 var 的后续赋值
var u: int32 = 0
u = 42
writeln(int32, u)
:quit

; 8. 控制流是 REPL 能力限制，不是语言 SyntaxError
if true then
while true then
goto done
label done:
break
continue
:help
:quit

; 9. 运行时失败的定义不污染已提交会话状态
var stable: int32 = 7
var failed: int32 = div(int32, 1, 0)
writeln(int32, stable)
writeln(int32, failed)
:quit
