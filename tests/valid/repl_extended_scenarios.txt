; REPL 扩展场景
; 1. 多行变量链式运算
var a: int32 = 10
var b: int32 = 20
var c: int32 = add(int32, a, b)
var d: int32 = mul(int32, c, 2)
writeln(int32, d)
:quit

; 2. 使用 let 常量再做运算
let N: int32 = 100
var x: int32 = add(int32, N, 50)
writeln(int32, x)
:quit

; 3. :reset 后重算
var z: int32 = 99
:reset
var z2: int32 = 11
var z3: int32 = 22
var z4: int32 = add(int32, z2, z3)
writeln(int32, z4)
:quit

; 4. 多进制字面量
var a: uint32 = 0xFF
var b: uint32 = 0b1010
writeln(uint32, add(uint32, a, b))
:quit

; 5. :vars 检查
var p: int8 = 42
var q: uint64 = 123
:vars
:quit

; 6. 类型边界
var a: int8 = 127
var b: int8 = 1
var c: int8 = add(int8, wrap, a, b)
writeln(int8, c)
:quit

; 7. 赋值操作给未初始化 var
var u: int32
u = 42
writeln(int32, u)
:quit
